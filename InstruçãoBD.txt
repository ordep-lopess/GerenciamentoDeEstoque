CREATE DATABASE bancoo;

USE bancoo;

-- cria tabela de produtos (onde ficam as doações)
CREATE TABLE IF NOT EXISTS produto (
  id INT(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  nome_doador VARCHAR(100) NOT NULL, 
  telefone VARCHAR(20) NOT NULL,
  email VARCHAR(100) NOT NULL,
  descricao VARCHAR(100) NOT NULL, 
  marca VARCHAR(50) NOT NULL,
  quantidade DECIMAL(8,2) NOT NULL,
  animal ENUM('cao','gato','aves') NOT NULL,
  tipo ENUM('racao','petisco','graos') NOT NULL,
  pacote_fechado ENUM('sim','nao') NOT NULL,
  data_doacao DATE NOT NULL
);

-- cria tabela de usuários (coisa básica)
CREATE TABLE IF NOT EXISTS usuarios (
  id INT(11) NOT NULL AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(150) NOT NULL,
  senha VARCHAR(70) NOT NULL,
  rua VARCHAR(100),
  num VARCHAR(20),
  cep VARCHAR(20)
);

-- tabela de movimentação (entradas/saídas)
-- produto_id pode ficar NULL depois que o produto for deletado
-- produto_nome guarda o nome quando o produto some
CREATE TABLE IF NOT EXISTS movimentacao (
  id INT AUTO_INCREMENT PRIMARY KEY,
  produto_id INT NULL,
  produto_nome VARCHAR(100) DEFAULT NULL, -- snapshot do nome
  tipo VARCHAR(20) NOT NULL, -- ex: entrada/saida
  quantidade DOUBLE NOT NULL,
  responsavel VARCHAR(150),
  observacao VARCHAR(500),
  data_movimentacao DATE NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- se não tiver a coluna produto_nome, tenta adicionar (verifica primeiro)
SET @has_col := (
  SELECT COUNT(*) FROM information_schema.COLUMNS
  WHERE TABLE_SCHEMA = 'bancoo' AND TABLE_NAME = 'movimentacao' AND COLUMN_NAME = 'produto_nome'
);
SET @sql_add_col = 'ALTER TABLE movimentacao ADD COLUMN produto_nome VARCHAR(100) DEFAULT NULL';
IF @has_col = 0 THEN
  PREPARE stmt_add_col FROM @sql_add_col;
  EXECUTE stmt_add_col;
  DEALLOCATE PREPARE stmt_add_col;
END IF;

-- garante que produto_id aceita NULL (só pra não dar ruim)
ALTER TABLE movimentacao MODIFY COLUMN produto_id INT NULL;

-- pega o nome da FK se houver, e dropa pra recriar
SET @fk_name := (
  SELECT CONSTRAINT_NAME
  FROM information_schema.KEY_COLUMN_USAGE
  WHERE TABLE_SCHEMA = 'bancoo'
    AND TABLE_NAME = 'movimentacao'
    AND COLUMN_NAME = 'produto_id'
    AND REFERENCED_TABLE_NAME = 'produto'
  LIMIT 1
);

IF @fk_name IS NOT NULL THEN
  SET @sql_drop_fk = CONCAT('ALTER TABLE movimentacao DROP FOREIGN KEY ', @fk_name);
  PREPARE stmt_drop_fk FROM @sql_drop_fk;
  EXECUTE stmt_drop_fk;
  DEALLOCATE PREPARE stmt_drop_fk;
END IF;

-- cria a FK de novo, agora dizendo pra zerar o produto_id quando o produto for apagado
ALTER TABLE movimentacao
  ADD CONSTRAINT fk_mov_produto FOREIGN KEY (produto_id) REFERENCES produto(id) ON DELETE SET NULL ON UPDATE CASCADE;

-- trigger bem simples: antes de apagar um produto, salva o nome na movimentacao e tira o id
DROP TRIGGER IF EXISTS trg_produto_before_delete;
DELIMITER $$
CREATE TRIGGER trg_produto_before_delete
BEFORE DELETE ON produto
FOR EACH ROW
BEGIN
  UPDATE movimentacao
    SET produto_nome = OLD.descricao,
        produto_id = NULL
  WHERE produto_id = OLD.id;
END$$
DELIMITER ;
